cmake_minimum_required(VERSION 3.16)
project(BabelMesh VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
option(BABELMESH_BUILD_GUI "Build the Qt 6 desktop CAD and meshing application" ON)
option(BABELMESH_BUILD_TESTS "Build solver-reader integration tests" ON)
set(BABELSIM_ROOT "${CMAKE_SOURCE_DIR}/../BabelSim" CACHE PATH "BabelSim checkout for real mesh-reader validation")

find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_path(OCCT_INCLUDE_DIR TopoDS_Shape.hxx PATH_SUFFIXES opencascade REQUIRED)
find_path(GMSH_INCLUDE_DIR gmsh.h HINTS "${CMAKE_SOURCE_DIR}/.cache/dev-packages/gmsh/usr/include" REQUIRED)
find_library(GMSH_LIBRARY NAMES gmsh)
if(NOT GMSH_LIBRARY)
  find_file(GMSH_LIBRARY NAMES libgmsh.so.4.8 PATHS /usr/lib/x86_64-linux-gnu /lib/x86_64-linux-gnu REQUIRED)
endif()
find_path(CGNS_INCLUDE_DIR cgnslib.h HINTS "${CMAKE_SOURCE_DIR}/.cache/dev-packages/cgns/usr/include" REQUIRED)
find_library(CGNS_LIBRARY NAMES cgns HINTS "${CMAKE_SOURCE_DIR}/.cache/dev-packages/cgns/usr/lib/x86_64-linux-gnu" REQUIRED)
set(OCCT_LIBRARIES)
foreach(component TKXDESTEP TKSTEP TKSTEP209 TKSTEPAttr TKSTEPBase TKXSBase TKXCAF TKCAF TKLCAF TKCDF TKV3d TKService TKOffset TKBO TKBool TKShHealing TKTopAlgo TKGeomAlgo TKGeomBase TKMesh TKPrim TKBRep TKG3d TKG2d TKMath TKernel)
  find_library(OCCT_${component} NAMES ${component} REQUIRED)
  list(APPEND OCCT_LIBRARIES ${OCCT_${component}})
endforeach()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/cad/*.cpp)
add_library(babelmesh-core STATIC ${CORE_SOURCES})
target_include_directories(babelmesh-core PUBLIC include ${OCCT_INCLUDE_DIR} PRIVATE ${GMSH_INCLUDE_DIR} ${CGNS_INCLUDE_DIR})
target_link_libraries(babelmesh-core PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json ${OCCT_LIBRARIES} ${GMSH_LIBRARY} ${CGNS_LIBRARY} pthread)
target_compile_options(babelmesh-core PRIVATE -Wall -Wextra -Wpedantic)
file(GLOB CLI_SOURCES CONFIGURE_DEPENDS src/cli/*.cpp)
add_executable(babelmesh ${CLI_SOURCES})
target_link_libraries(babelmesh PRIVATE babelmesh-core)
if(BABELMESH_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets OpenGLWidgets)
  add_executable(babelmesh-gui src/gui/main.cpp src/gui/studio.cpp src/gui/mesh_view.cpp src/gui/icons.qrc)
  set_target_properties(babelmesh-gui PROPERTIES AUTORCC ON)
  target_link_libraries(babelmesh-gui PRIVATE Qt6::Widgets Qt6::OpenGLWidgets babelmesh-core)
  target_compile_definitions(babelmesh-gui PRIVATE BABELMESH_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  target_compile_options(babelmesh-gui PRIVATE -Wall -Wextra -Wpedantic)
  add_dependencies(babelmesh-gui babelmesh)
endif()
if(BABELMESH_BUILD_TESTS)
  enable_testing()
  add_executable(mixed_mesh_test tests/mixed_mesh_test.cpp ${BABELSIM_ROOT}/src/core/mesh.cpp ${BABELSIM_ROOT}/src/io/mesh_reader.cpp)
  target_include_directories(mixed_mesh_test PRIVATE ${BABELSIM_ROOT}/include)
  target_link_libraries(mixed_mesh_test PRIVATE babelmesh-core)
  add_test(NAME mixed-mesh-contract COMMAND mixed_mesh_test ${CMAKE_SOURCE_DIR})
  add_executable(babelsim-mesh-check tools/m4_solver_quality_probe.cpp ${BABELSIM_ROOT}/src/core/mesh.cpp ${BABELSIM_ROOT}/src/io/mesh_reader.cpp)
  target_include_directories(babelsim-mesh-check PRIVATE ${BABELSIM_ROOT}/include)
  target_link_libraries(babelsim-mesh-check PRIVATE nlohmann_json::nlohmann_json)
endif()
